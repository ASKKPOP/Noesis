# Phase 38 Plan 04: WSS subscription + per-DID firehose filter

## Objective

Open the inbound channel. Brain subscribes to the Grid firehose over WSS with the same EdDSA bearer used in REST, and Grid filters frames per subscriber Civic-DID so each Brain only sees events relevant to its Nous. This finishes WIRE-01 (the WSS half) and ships WIRE-05.

The filter is server-side and applied at egress (inside `ClientConnection.trySend`), so the audit chain itself is untouched. R-31-01 zero-diff generalizes across the network boundary: the chain-head hash is identical whether one Brain is connected or fifty.

## Context

`WsFirehoseHub` already accepts a `DIDContext` per client (Phase 36 VIS-03). `ClientConnection.trySend` branches on `didContext?.tier === 'civic_member'` and chooses `serializeFullFrame` vs `serializeVisitorFrame`. For WIRE-05, the only addition is an EARLIER short-circuit: if `didContext?.tier === 'civic_member'` AND `!isRelevantFor(entry, didContext)`, drop the frame for THIS client only. Visitor (anonymous / human_visitor) behavior is unchanged — they still see redacted everything.

The relevance predicate per user decision D-38-A3:

```ts
function isRelevantFor(entry: AuditEntry, didContext: DIDContext | null): boolean {
    // Visitors (no DID or non-civic) bypass — Phase 36 VIS-03 redaction handles them.
    if (!didContext || didContext.tier !== 'civic_member') return true;
    const did = didContext.did;
    // 1. Own actor — every audit echo of MY actions.
    if (entry.actorDid === did) return true;
    // 2. Targeted at me — message I'm meant to receive.
    if (entry.targetDid === did) return true;
    // 3. Community events — Phase 38 has no membership tables, so filter by actor only.
    //    TODO Phase 49: extend to subscribers' community memberships via community_members table.
    if (entry.eventType.startsWith('community.') && entry.actorDid === did) return true;
    // 4. World heartbeat — every subscriber sees ticks.
    if (entry.eventType === 'tick') return true;
    return false;
}
```

The Brain-side subscriber uses `websockets>=12.0` (added in Plan 38-01's pyproject change). Exponential backoff with jitter prevents the thundering herd on Grid restart (Pitfall 3).

   ```ts
   import type { FastifyInstance } from 'fastify';
   import type { WsFirehoseHub } from '../../audit/firehose-hub.js';
   import { tryDid } from '../preHandlers/tryDid.js';

   type Services = {
       didStore?: { isRevoked(did: string): boolean | Promise<boolean> };
       brainTokenStore?: { /* from Plan 38-01 */ };
   };

   export function registerBrainFirehoseRoute(
       instance: FastifyInstance,
       firehoseHub: WsFirehoseHub,
       services?: Services,
   ): void {
       instance.get('/api/v1/brain/firehose', { websocket: true }, async (socket, req) => {
           // Resolve DIDContext from Authorization bearer. tryDid (Plan 38-02) handles
           // EdDSA Brain JWTs via brain_tokens lookup.
           const didContext = await tryDid(req, {
               didStore: services?.didStore,
               brainTokenStore: services?.brainTokenStore,
           });
           if (!didContext || didContext.tier !== 'civic_member') {
               try { socket.close(4401, 'civic_did_required'); } catch { /* swallow */ }
               return;
           }

           // ServerSocket adapter — same shape as audit-firehose.ts.
           const adapter = {
               get bufferedAmount() { return socket.bufferedAmount; },
               send: (data: string) => socket.send(data),
               close: (code?: number, reason?: string) => socket.close(code, reason),
               on: (event: 'message' | 'close' | 'error', cb: (arg: never) => void) => {
                   if (event === 'message') {
                       socket.on('message', (data: Buffer) =>
                           (cb as (d: unknown) => void)(data.toString('utf8')));
                   } else if (event === 'close') {
                       socket.on('close', () => (cb as () => void)());
                   } else {
                       socket.on('error', (err: Error) =>
                           (cb as (e: Error) => void)(err));
                   }
               },
           };

           firehoseHub.onConnect(adapter, didContext);
       });
   }
   ```

1. Implement `WssSubscriber`:

   ```python
   from __future__ import annotations
   import asyncio
   import json
   import logging
   import random
   from typing import Awaitable, Callable, Optional
   import websockets
   from .token_manager import TokenManager

   log = logging.getLogger(__name__)

   BASE_BACKOFF_SECONDS = 1.0
   MAX_BACKOFF_SECONDS = 60.0

   class WssSubscriber:
       def __init__(
           self,
           *,
           grid_url: str,                      # https:// — we rewrite to wss://
           token_manager: TokenManager,
           on_frame: Callable[[dict], Awaitable[None]],
       ) -> None:
           # Accept both https/wss forms — internally always wss.
           if grid_url.startswith("https://"):
               base = "wss://" + grid_url[len("https://"):]
           elif grid_url.startswith("wss://"):
               base = grid_url
           else:
               raise ValueError(f"WssSubscriber requires https:// or wss:// URL, got {grid_url!r}")
           self._url = f"{base.rstrip('/')}/api/v1/brain/firehose"
           self._token_manager = token_manager
           self._on_frame = on_frame
           self._stop = asyncio.Event()
           self._task: Optional[asyncio.Task] = None

       async def start(self) -> None:
           if self._task is not None:
               raise RuntimeError("WssSubscriber already started")
           self._task = asyncio.create_task(self._run_forever())

       async def stop(self) -> None:
           self._stop.set()
           if self._task is not None:
               try:
                   await asyncio.wait_for(self._task, timeout=5.0)
               except asyncio.TimeoutError:
                   self._task.cancel()
               self._task = None

       async def _run_forever(self) -> None:
           attempt = 0
           while not self._stop.is_set():
               try:
                   await self._connect_once()
                   attempt = 0  # successful session → reset backoff
               except Exception as exc:
                   log.warning("[Brain] WSS connection lost: %s", exc)
               if self._stop.is_set():
                   return
               # Exponential backoff with jitter (Pitfall 3).
               delay = min(MAX_BACKOFF_SECONDS, BASE_BACKOFF_SECONDS * (2 ** attempt))
               jitter = 0.75 + 0.5 * random.random()  # [0.75, 1.25]
               sleep_for = delay * jitter
               try:
                   await asyncio.wait_for(self._stop.wait(), timeout=sleep_for)
                   return  # stop signaled during backoff
               except asyncio.TimeoutError:
                   pass
               attempt = min(attempt + 1, 6)  # cap exponent so delay <= 60s

       async def _connect_once(self) -> None:
           token = self._token_manager.get_valid_token()
           headers = {"Authorization": f"Bearer {token}"}
           async with websockets.connect(self._url, extra_headers=headers) as ws:
               log.info("[Brain] WSS connected to %s", self._url)
               while not self._stop.is_set():
                   raw = await ws.recv()
                   try:
                       frame = json.loads(raw)
                   except json.JSONDecodeError:
                       log.warning("[Brain] WSS dropped non-JSON frame: %r", raw[:80])
                       continue
                   try:
                       await self._on_frame(frame)
                   except Exception:
                       log.exception("[Brain] WSS on_frame handler raised")
   ```
