```typescript
async function handleForkConfirm() {
    setForkOpen(false);
    setForkStatus('loading');
    setForkError(null);
    try {
        const res = await fetch(`/api/v1/operator/fork/${encodeURIComponent(civicDid)}`, {
            method: 'POST',
            credentials: 'include',
            headers: {
                'Content-Type': 'application/json',
                // x-operator-tier and x-operator-id are server-injected via Steward middleware
                // OR set explicitly if Steward client-side has them in context:
                // 'x-operator-tier': '4',
                // 'x-operator-id': operatorId,
            },
        });
        if (!res.ok) {
            const body = await res.json().catch(() => ({error: 'unknown'}));
            const msg = body.detail || body.error || `HTTP ${res.status}`;
            setForkError(msg);
            setForkStatus('error');
            return;
        }
        const data = await res.json();
        setForkResult(data);
        setForkStatus('success');
        // Trigger download: programmatic <a download> click
        const a = document.createElement('a');
        a.href = data.download_url;
        a.download = '';  // browser uses Content-Disposition filename
        document.body.appendChild(a);
        a.click();
        document.body.removeChild(a);
    } catch (err) {
        setForkError(String(err));
        setForkStatus('error');
    }
}
```
