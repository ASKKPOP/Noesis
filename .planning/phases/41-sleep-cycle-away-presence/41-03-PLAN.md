**File 1: `grid/src/civic-presence/presence-service.ts`** — facade that composes Plan 02 stores. This is the dependency-injection seam consumed by `launcher.ts`, `firehose-hub.ts`, and route handlers (Plan 04).

```typescript
/**
 * Phase 41 — PresenceService facade.
 *
 * Composes PresenceStore + MessageQueueStore + GraceTimerRegistry + CivicDidStore +
 * BusinessDidStore + AuditChain into a single dependency-injection surface for routes
 * and lifecycle hooks (firehose-hub, launcher).
 */

import type { AuditChain } from '../audit/chain.js';
import type { CivicDidStore } from '../civic-registry/civic-did-store.js';
import type { BusinessDidStore } from '../civic-registry/business-did-store.js';
import { GRACE_TIMER_MS } from './types.js';
import type { PresenceStatus, PresenceRecord, QueuedMessage } from './types.js';
import type { GraceTimerRegistry } from './grace-timer-registry.js';
import type { PresenceStore } from './presence-store.js';
import type { MessageQueueStore } from './message-queue-store.js';
import { runEscalationCheck } from './escalation-check.js';

export interface PresenceServiceDeps {
    readonly gridName: string;
    readonly presenceStore: PresenceStore;
    readonly messageQueueStore: MessageQueueStore;
    readonly graceTimerRegistry: GraceTimerRegistry;
    readonly civicDidStore: CivicDidStore;
    readonly businessDidStore?: BusinessDidStore;
    readonly audit: AuditChain;
    /** Callback returning the current tick (avoids holding a reference to WorldClock). */
    readonly currentTick: () => number;
}

export class PresenceService {
    constructor(private readonly deps: PresenceServiceDeps) {}

    /** SLEEP-01 — WSS connection opened for a civic_member. */
    onWsConnect(civicDid: string): void {
        this.deps.graceTimerRegistry.incrementConnection(civicDid);
    }

    /**
     * SLEEP-01 — WSS connection closed. When refcount reaches 0, starts grace timer.
     * Timer callback flips status to 'away' if no heartbeat/reconnect within GRACE_TIMER_MS.
     */
    onWsDisconnect(civicDid: string): void {
        const tick = this.deps.currentTick();
        this.deps.graceTimerRegistry.decrementConnection(
            civicDid,
            () => {
                // Fire-and-forget — never throw into the event loop.
                void this.deps.presenceStore
                    .updatePresence(this.deps.gridName, civicDid, 'away', tick)
                    .catch(() => { /* swallowed; structured log left to caller wrapper */ });
            },
            GRACE_TIMER_MS,
        );
    }

    /** SLEEP-01 / D-41-02 — heartbeat resets timer + writes last_seen_at=NOW + last_seen_tick. */
    async onHeartbeat(civicDid: string, tick: number): Promise<PresenceRecord | null> {
        this.deps.graceTimerRegistry.heartbeat(civicDid);
        await this.deps.presenceStore.updatePresence(this.deps.gridName, civicDid, 'awake', tick);
        return this.deps.presenceStore.getPresence(this.deps.gridName, civicDid);
    }

    /** SLEEP-02 — POST /civic/message backing. Returns null if queue full (T-41-02). */
    async sendDirectMessage(
        recipientCivicDid: string,
        senderCivicDid: string,
        messageJson: unknown,
    ): Promise<{ queued: boolean; messageId: number | null; queueFull: boolean }> {
        const recipientPresence = await this.deps.presenceStore.getPresence(
            this.deps.gridName, recipientCivicDid,
        );
        if (!recipientPresence) {
            return { queued: false, messageId: null, queueFull: false };
        }
        if (recipientPresence.presenceStatus === 'awake') {
            // Recipient online — Phase 41 still enqueues for simplicity; Brain pulls via inbox.
            // (Direct push via WSS is Phase 42 P2P concern.)
        }
        const tick = this.deps.currentTick();
        const messageId = await this.deps.messageQueueStore.enqueue(
            this.deps.gridName, recipientCivicDid, senderCivicDid, messageJson, tick,
        );
        if (messageId === null) {
            return { queued: false, messageId: null, queueFull: true };
        }
        return { queued: true, messageId, queueFull: false };
    }

    /** SLEEP-03 — GET /civic/inbox backing. */
    async listInbox(civicDid: string, sinceTick: number | undefined): Promise<{
        messages: QueuedMessage[];
        queueDepth: number;
    }> {
        const [messages, queueDepth] = await Promise.all([
            this.deps.messageQueueStore.listPending(this.deps.gridName, civicDid, sinceTick),
            this.deps.messageQueueStore.depth(this.deps.gridName, civicDid),
        ]);
        return { messages, queueDepth };
    }

    /** SLEEP-03 — PATCH /civic/inbox/ack backing. */
    async ackMessages(civicDid: string, messageIds: number[]): Promise<number> {
        return this.deps.messageQueueStore.markDelivered(this.deps.gridName, civicDid, messageIds);
    }

    /** T-41-04 mitigation — surface for requireCivicDid preHandler (Plan 04). */
    async isFrozen(civicDid: string): Promise<boolean> {
        return this.deps.presenceStore.isFrozen(this.deps.gridName, civicDid);
    }

    /** Public Civic Map polling — Plan 04 GET /api/v1/civic/presence backing. */
    async listAllPresence(): Promise<PresenceRecord[]> {
        return this.deps.presenceStore.listAll(this.deps.gridName);
    }

    /** SLEEP-03 fallback — Plan 04 GET /api/v1/civic/presence/me backing. */
    async getPresence(civicDid: string): Promise<PresenceRecord | null> {
        return this.deps.presenceStore.getPresence(this.deps.gridName, civicDid);
    }

    /** SLEEP-04 + SLEEP-05 — 24h escalation walker. Called by launcher setInterval. */
    async runEscalationCheck(now: Date, tick: number): Promise<{
        absentCount: number;
        presumedDepartedCount: number;
    }> {
        return runEscalationCheck({
            gridName: this.deps.gridName,
            presenceStore: this.deps.presenceStore,
            messageQueueStore: this.deps.messageQueueStore,
            civicDidStore: this.deps.civicDidStore,
            businessDidStore: this.deps.businessDidStore,
            audit: this.deps.audit,
            now,
            tick,
        });
    }
}
```

**File 2: `grid/src/civic-presence/escalation-check.ts`** — pure function that walks stale rows and applies transitions:

```typescript
/**
 * Phase 41 / SLEEP-04 + SLEEP-05 — 24h escalation walker.
 *
 * - 30d offline → status='absent'; enqueue self-message from 'did:civic:noesis:system'
 *   (Phase 49 stub — community charter revocation processing deferred).
 * - 1y offline  → status='presumed_departed'; markFrozen on CivicDidStore;
 *   markDissolved on BusinessDidStore (if linked); appendIrsDisbursementExecuted
 *   (audit-chain only — Phase 41 broadcast allowlist delta = 0).
 */

import type { AuditChain } from '../audit/chain.js';
import type { CivicDidStore } from '../civic-registry/civic-did-store.js';
import type { BusinessDidStore } from '../civic-registry/business-did-store.js';
import { appendIrsDisbursementExecuted } from '../audit/append-irs-disbursement-executed.js';
import type { PresenceStore } from './presence-store.js';
import type { MessageQueueStore } from './message-queue-store.js';
import { ABSENT_THRESHOLD_MS, PRESUMED_DEPARTED_THRESHOLD_MS } from './types.js';

const SYSTEM_SENDER_CIVIC_DID = 'did:civic:noesis:system';

export interface EscalationDeps {
    readonly gridName: string;
    readonly presenceStore: PresenceStore;
    readonly messageQueueStore: MessageQueueStore;
    readonly civicDidStore: CivicDidStore;
    readonly businessDidStore?: BusinessDidStore;
    readonly audit: AuditChain;
    readonly now: Date;
    readonly tick: number;
}

export async function runEscalationCheck(deps: EscalationDeps): Promise<{
    absentCount: number;
    presumedDepartedCount: number;
}> {
    const absentBoundary = new Date(deps.now.getTime() - ABSENT_THRESHOLD_MS);
    const presumedBoundary = new Date(deps.now.getTime() - PRESUMED_DEPARTED_THRESHOLD_MS);

    // SLEEP-05: 1y boundary FIRST (these are also stale by 30d, but we want
    // them flipped to presumed_departed, not absent).
    const presumed = await deps.presenceStore.listStaleSince(
        deps.gridName,
        presumedBoundary,
        ['away', 'absent'],
    );
    for (const row of presumed) {
        await deps.presenceStore.updatePresence(
            deps.gridName, row.civicDid, 'presumed_departed', deps.tick,
        );
        await deps.civicDidStore.markFrozen(deps.gridName, row.civicDid, deps.tick);
        // BusinessDid dissolution — only if linked. Use the existing markDissolved method.
        // If the executor finds the BusinessDidStore method name differs (e.g. dissolve, markDissolved, revokeAll),
        // adapt accordingly. The README of business-did-store.ts is authoritative.
        if (deps.businessDidStore) {
            try {
                // Best-effort — if no business-did is linked to this civic-did, store
                // should handle gracefully. Phase 41 does not enumerate businesses;
                // Phase 44 will do exhaustive cleanup. For now, fire-and-forget.
                // @ts-ignore — markDissolved signature confirmed via grep during implementation
                await deps.businessDidStore.markDissolvedByCivicDid?.(
                    deps.gridName, row.civicDid, deps.tick,
                );
            } catch { /* swallow — best-effort */ }
        }
        // SLEEP-05 audit chain entry (NOT broadcast — allowlist delta = 0 for Phase 41).
        appendIrsDisbursementExecuted(deps.audit, {
            amount_bios: 0, // Phase 41 stub — actual Bios accounting in Phase 45.
            cause: 'presumed_departed',
            civic_did: row.civicDid,
            grid_name: deps.gridName,
            tick: deps.tick,
        });
    }

    // SLEEP-04: 30d boundary — only rows still in 'away' AFTER the 1y pass.
    // Note: rows that were just flipped to presumed_departed above won't match
    // the 'away' status filter on this second pass.
    const absent = await deps.presenceStore.listStaleSince(
        deps.gridName,
        absentBoundary,
        ['away'],
    );
    for (const row of absent) {
        await deps.presenceStore.updatePresence(
            deps.gridName, row.civicDid, 'absent', deps.tick,
        );
        // Phase 49 stub — community charter revocation queue. We enqueue a self-message
        // from the system sender into the Nous's own civic_message_queue. Brain sees
        // it on reconnect. Real charter processing lands in Phase 49.
        await deps.messageQueueStore.enqueue(
            deps.gridName,
            row.civicDid,
            SYSTEM_SENDER_CIVIC_DID,
            {
                kind: 'absent_escalation_notice',
                message: 'You were marked absent due to 30+ days offline. ' +
                         'Community charter revocation processing is pending (Phase 49 stub).',
                escalated_at_tick: deps.tick,
            },
            deps.tick,
        );
    }

    return {
        absentCount: absent.length,
        presumedDepartedCount: presumed.length,
    };
}
```
