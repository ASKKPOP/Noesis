    import { describe, it, expect, beforeAll, afterAll } from 'vitest';
    import mysql from 'mysql2/promise';
    import { MigrationRunner } from '../../src/db/migration-runner.js';
    import { MIGRATIONS } from '../../src/db/schema.js';

    // Uses the same test MySQL DB as other grid integration tests.
    // If no test DB is available in CI, skip with .skipIf(process.env.CI === 'true').
    // Check how existing integration tests connect to MySQL.

    describe('Norm migration (version 7)', () => {
        let pool: mysql.Pool;

        beforeAll(async () => {
            // Connect to test DB — use same env vars as other grid integration tests
            pool = mysql.createPool({
                host: process.env.MYSQL_HOST ?? 'localhost',
                port: Number(process.env.MYSQL_PORT ?? 3306),
                user: process.env.MYSQL_USER ?? 'root',
                password: process.env.MYSQL_PASSWORD ?? '',
                database: process.env.MYSQL_DATABASE ?? 'noesis_test',
                multipleStatements: true,
            });
            const runner = new MigrationRunner(MIGRATIONS);
            await runner.run(pool);
        });

        afterAll(async () => {
            await pool.end();
        });

        it('norm_candidates table exists with correct columns', async () => {
            const [rows] = await pool.query(
                "SELECT COLUMN_NAME, DATA_TYPE FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'norm_candidates' ORDER BY ORDINAL_POSITION"
            ) as [Array<{COLUMN_NAME: string; DATA_TYPE: string}>, unknown];
            const cols = rows.map(r => r.COLUMN_NAME);
            expect(cols).toContain('fingerprint');
            expect(cols).toContain('grid_name');
            expect(cols).toContain('participant_dids');
            expect(cols).toContain('first_seen_tick');
            expect(cols).toContain('last_updated_tick');
        });

        it('norm_registry table exists with correct columns', async () => {
            const [rows] = await pool.query(
                "SELECT COLUMN_NAME, DATA_TYPE FROM INFORMATION_SCHEMA.COLUMNS WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'norm_registry' ORDER BY ORDINAL_POSITION"
            ) as [Array<{COLUMN_NAME: string; DATA_TYPE: string}>, unknown];
            const cols = rows.map(r => r.COLUMN_NAME);
            expect(cols).toContain('norm_id');
            expect(cols).toContain('fingerprint');
            expect(cols).toContain('crystallized_tick');
            expect(cols).toContain('participant_count');
            expect(cols).toContain('convergence_type');
            expect(cols).toContain('event_hash');
            expect(cols).toContain('grid_name');
        });

        it('norm_candidates primary key is (fingerprint, grid_name)', async () => {
            const [rows] = await pool.query(
                "SELECT COLUMN_NAME FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'norm_candidates' AND CONSTRAINT_NAME = 'PRIMARY' ORDER BY ORDINAL_POSITION"
            ) as [Array<{COLUMN_NAME: string}>, unknown];
            const pkCols = rows.map(r => r.COLUMN_NAME);
            expect(pkCols).toEqual(['fingerprint', 'grid_name']);
        });

        it('norm_registry primary key is norm_id', async () => {
            const [rows] = await pool.query(

            ) as [Array<{COLUMN_NAME: string}>, unknown];
            expect(rows[0]?.COLUMN_NAME).toBe('norm_id');
        });

        it('migration is idempotent (running twice does not error)', async () => {
            const runner = new MigrationRunner(MIGRATIONS);
