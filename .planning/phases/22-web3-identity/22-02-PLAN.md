**File 1 — `grid/src/api/portal/auth.ts`** (new file):

```typescript
/**
 * Portal auth routes — SIWE authentication, JWT issuance.
 *
 * Phase 22 WEB3-01 through WEB3-06.
 *
 * Routes:
 *   GET  /api/v1/portal/auth/nonce   — fresh nonce (5-min TTL)
 *   POST /api/v1/portal/auth/verify  — SIWE verify → JWT cookie
 *   POST /api/v1/portal/auth/logout  — clear JWT cookie
 *   GET  /api/v1/portal/auth/me      — current user from JWT
 */

import { createHash, randomUUID } from 'node:crypto';
import { SiweMessage } from 'siwe';
import { SignJWT, jwtVerify, generateKeyPair } from 'jose';
import type { FastifyInstance } from 'fastify';
import type { GridServices } from '../server.js';
import { appendHumanJoined } from '../../audit/append-human-joined.js';

const COOKIE_NAME = 'noesis_portal_token';
const NONCE_TTL_MS = 5 * 60 * 1000; // 5 minutes

/** In-memory nonce store. Key = nonce string, Value = created timestamp. */
const nonceMap = new Map<string, number>();

/** ES256 key pair — generated once at module load. */
const keyPairPromise = generateKeyPair('ES256');

export function registerPortalAuthRoutes(
    app: FastifyInstance,
    services: GridServices,
): void {
    // GET /api/v1/portal/auth/nonce
    app.get('/api/v1/portal/auth/nonce', async (_req, reply) => {
        const nonce = randomUUID();
        nonceMap.set(nonce, Date.now());
        return reply.send({ nonce });
    });

    // POST /api/v1/portal/auth/verify
    app.post<{
        Body: { message: unknown; signature: unknown };
    }>('/api/v1/portal/auth/verify', async (req, reply) => {
        if (!services.humanRegistry) {
            return reply.status(503).send({ error: 'human_registry_unavailable' });
        }

        const { message, signature } = req.body ?? {};
        if (typeof message !== 'object' || message === null || typeof signature !== 'string') {
            return reply.status(400).send({ error: 'invalid_request' });
        }

        let siweMessage: SiweMessage;
        try {
            siweMessage = new SiweMessage(message as Record<string, unknown>);
        } catch {
            return reply.status(400).send({ error: 'invalid_siwe_message' });
        }

        const nonce = siweMessage.nonce;
        const nonceCreatedAt = nonceMap.get(nonce);
        if (nonceCreatedAt === undefined) {
            return reply.status(401).send({ error: 'nonce_unknown' });
        }
        if (Date.now() - nonceCreatedAt > NONCE_TTL_MS) {
            nonceMap.delete(nonce);
            return reply.status(401).send({ error: 'nonce_expired' });
        }
        nonceMap.delete(nonce);

        let verifyResult: Awaited<ReturnType<typeof siweMessage.verify>>;
        try {
            verifyResult = await siweMessage.verify({ signature, nonce });
        } catch {
            return reply.status(401).send({ error: 'invalid_signature' });
        }
        if (!verifyResult.success) {
            return reply.status(401).send({ error: 'invalid_signature' });
        }

        const ethAddress = siweMessage.address; // checksummed from SIWE
        const gridName = services.gridName;

        // Find or create human record.
        let human = services.humanRegistry.findByAddress(gridName, ethAddress);
        const isNew = human === undefined;
        if (!human) {
            human = services.humanRegistry.createHuman({ eth_address: ethAddress, grid_name: gridName });

            // WEB3-04: human.joined audit event on first creation only.
            const eth_address_hash = createHash('sha256')
                .update(ethAddress.toLowerCase())
                .digest('hex');
            // services.audit may not be present in tests — guard it.
            if (services.audit) {
                appendHumanJoined(services.audit, {
                    human_did: human.did,
                    eth_address_hash,
                    grid_name: gridName,
                    tick: services.clock.currentTick ?? 0,
                });
            }
        }

        // Issue JWT (ES256, 24h).
        const { privateKey } = await keyPairPromise;
        const token = await new SignJWT({ did: human.did, eth_address: human.eth_address, grid_name: gridName })
            .setProtectedHeader({ alg: 'ES256' })
            .setIssuedAt()
            .setExpirationTime('24h')
            .sign(privateKey);

        reply.setCookie(COOKIE_NAME, token, {
            httpOnly: true,
            sameSite: 'strict',
            secure: process.env['NODE_ENV'] === 'production',
            path: '/',
            maxAge: 24 * 60 * 60, // seconds
        });

        return reply.send({ did: human.did, eth_address: human.eth_address, is_new: isNew });
    });

    // POST /api/v1/portal/auth/logout
    app.post('/api/v1/portal/auth/logout', async (_req, reply) => {
        reply.clearCookie(COOKIE_NAME, { path: '/' });
        return reply.send({ ok: true });
    });

    // GET /api/v1/portal/auth/me
    app.get('/api/v1/portal/auth/me', async (req, reply) => {
        const token = req.cookies?.[COOKIE_NAME];
        if (!token) {
            return reply.status(401).send({ error: 'not_authenticated' });
        }
        try {
            const { publicKey } = await keyPairPromise;
            const { payload } = await jwtVerify(token, publicKey);
            return reply.send({
                did: payload['did'],
                eth_address: payload['eth_address'],
            });
        } catch {
            return reply.status(401).send({ error: 'invalid_token' });
        }
    });
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → SIWE verify endpoint | Untrusted body: message (object) + signature (string) |
| JWT cookie → /me endpoint | Untrusted cookie value; must be verified with public key before trusting |
| ETH address → audit payload | Raw address from SIWE must be SHA-256 hashed before entering audit chain (WEB3-04) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-22-02-01 | Spoofing | POST /portal/auth/verify | mitigate | SiweMessage.verify cryptographically validates the signature against the address — no trust without valid secp256k1 signature |
| T-22-02-02 | Tampering | Nonce replay | mitigate | Nonce deleted from map immediately after first use; 5-min TTL prevents stale nonce reuse |
| T-22-02-03 | Information Disclosure | human.joined audit payload | mitigate | ETH address SHA-256 hashed before audit append per WEB3-04; raw address never in audit chain |
| T-22-02-04 | Elevation of Privilege | JWT cookie | mitigate | httpOnly cookie prevents JS access; sameSite strict prevents CSRF; ES256 asymmetric key prevents forgery |
| T-22-02-05 | Denial of Service | nonceMap growth | accept | Nonces are UUIDs with 5-min TTL; low-volume Portal usage in v2.5; Redis deferred to later phase per phase decisions |
| T-22-02-06 | Repudiation | human.joined double-fire | mitigate | findByAddress check before createHuman; human.joined only on isNew === true (WEB3-06) |
</threat_model>
