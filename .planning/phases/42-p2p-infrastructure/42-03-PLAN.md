```typescript
/**
 * Phase 42 P2P-05 — sole producer for p2p.peer_announced (allowlist position 65).
 * Closed 3-key payload {civic_did_hash, endpoint_hash, tick}. Keys ALPHABETICAL.
 * D-42-02: endpoint_hash = sha256('online') — static sentinel (no IP/port leakage).
 */
import type { AuditChain } from './chain.js';
import type { AuditEntry } from './types.js';
import { payloadPrivacyCheck } from './broadcast-allowlist.js';

const HEX64_RE = /^[0-9a-f]{64}$/i;

export interface P2pPeerAnnouncedPayload {
    readonly civic_did_hash: string;
    readonly endpoint_hash: string;
    readonly tick: number;
}
const EXPECTED_KEYS = ['civic_did_hash', 'endpoint_hash', 'tick'] as const;

export function appendP2pPeerAnnounced(
    audit: AuditChain,
    payload: P2pPeerAnnouncedPayload,
): AuditEntry {
    // 1. Type guard
    if (payload === null || typeof payload !== 'object' || Array.isArray(payload)) {
        throw new TypeError('appendP2pPeerAnnounced: payload must be a plain object');
    }
    // 2. Regex/format guards
    if (!HEX64_RE.test(payload.civic_did_hash)) {
        throw new TypeError(`appendP2pPeerAnnounced: civic_did_hash must match HEX64`);
    }
    if (!HEX64_RE.test(payload.endpoint_hash)) {
        throw new TypeError(`appendP2pPeerAnnounced: endpoint_hash must match HEX64`);
    }
    // 3. Integer guards
    if (!Number.isInteger(payload.tick) || payload.tick < 0) {
        throw new TypeError('appendP2pPeerAnnounced: tick must be non-negative integer');
    }
    // 5. Closed-tuple structural check (alphabetical)
    const keys = Object.keys(payload).sort();
    if (keys.length !== EXPECTED_KEYS.length || !EXPECTED_KEYS.every((k, i) => k === keys[i])) {
        throw new TypeError(
            `appendP2pPeerAnnounced: closed-tuple violation — expected ${JSON.stringify(EXPECTED_KEYS)} got ${JSON.stringify(keys)}`,
        );
    }
    // 6. Explicit reconstruction (no spread)
    const cleanPayload = {
        civic_did_hash: payload.civic_did_hash,
        endpoint_hash: payload.endpoint_hash,
        tick: payload.tick,
    };
    // 7. Privacy gate (belt-and-suspenders)
    const privacy = payloadPrivacyCheck(cleanPayload);
    if (!privacy.ok) {
        throw new TypeError(
            `appendP2pPeerAnnounced: privacy violation — path=${privacy.offendingPath} keyword=${privacy.offendingKeyword}`,
        );
    }
    // 8. Commit — actorDid is civic_did_hash (the announcing party)
    return audit.append('p2p.peer_announced', payload.civic_did_hash, cleanPayload);
}
```
